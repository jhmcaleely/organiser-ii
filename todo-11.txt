                                  TODO v1.1
                               
                                INSTRUCTIONS

TODO provides a sorted todo list, ordered by priority and due date.

RUN TODO to start it up.

When it first executes it creates A:TODOITEM to store the list in.

When the program is in use the screen looks like:

    +--------------------+
    |1 10d Do something  |
    |1     Do something else..        <-- An item with no due date
    |2  2w something less important
    |                    |
    +--------------------+
     ^  ^
     |  Days/Weeks until item due (negative if overdue). 
     |  The display becomes days if it is due sooner than 2 weeks
     |
     Priority: High=1, Medium=2, Low=3

The keys available are:

MODE

  This inserts a new item. You will be prompted for a description and a 
  priority. You will then be asked if there is a due date. If so you can then 
  pick it in the standard psion way. (Using a routine supplied by David C. 
  Jones) The entry is added to the list, but may not appear on screen until you
  scroll to its location.

DEL

  Deletes the entry on the top line of the screen.

UP and DOWN arrow

  Use to scroll around the list.

(*----------------------------------------------------------------------------*)

                                     FILES

Files created:

  A:TODOITEM	The todo list
  
  This file is maintained in sorted order by the program so A: is the only 
  practical drive for it.

		and to allow the user to escape (returns -1)

BATT		Check for battery low error

REM To Do list manager
REM (c) John McAleely 1994, except where stated in subroutines
REM 2nd APRIL 1994

LOCAL g%,p%

IF NOT EXIST("A:TODOITEM")
  CLS
  printc:("To Do List Manager",1)
  printc:(CHR$(184)+" J.H. McAleely 1994",2)
  printc:("(except where",3 )
  printc:("credited in source)",4)
  PAUSE -40
  KEY
  CREATE "A:TODOITEM",A,desc$,pr%,due
ELSE
  OPEN "A:TODOITEM",A,desc$,pr%,due
ENDIF
UDG 0,23,0,23,0,23,0,0,31
p%=1

DO
  
  g%=TDISP%:(p%)
  IF g%=2
    TADD:
  ELSEIF g%=3 AND p%>1
    p%=p%-1
  ELSEIF g%=4 AND p%<COUNT
    p%=p%+1
  ELSEIF g%=8 AND COUNT>0
    ERASE
    IF p%>COUNT
      p%=COUNT
    ENDIF
  ENDIF
UNTIL g%=1
(*----------------------------------------------------------------------------*)
TADD:
REM Adds a to do item in the correct sorted position
REM in the order priority,due date.

LOCAL d$(64),p%,d,g%,pos%,loop

CLS
topline:

DO
  AT 1,2
  PRINT "Description:",
  TRAP INPUT d$
  IF ERR=206
    RETURN
  ENDIF
UNTIL LEN(d$)<65
DO
  AT 1,3
  PRINT "Priority:"
  AT 1,4
  p%=MENUN(1,"high,medium,low")
  IF p%=0
    RETURN
  ENDIF
UNTIL p%>0 AND p%<4
CLS :topline:
AT 1,2
PRINT LEFT$(d$,20)
AT 1,3
PRINT "Due Date? (Y/N)"
g%=GET
IF g%=%y OR g%=%Y
  d=DATEIN:(1,3,DAY,MONTH,YEAR)
  IF d=-1
    RETURN
  ENDIF
ELSE
  d=99999
ENDIF

FIRST
WHILE NOT EOF
  IF a.pr%<p%
    NEXT
  ELSEIF a.pr%=p% AND a.due<=d
    NEXT
  ELSE
    BREAK
  ENDIF
ENDWH
pos%=POS
a.desc$=d$
a.pr%=p%
a.due=d
APPEND
IF NOT EOF
  loop=COUNT-1-pos%
  WHILE loop>=0
    POSITION pos%
    UPDATE
    loop=loop-1
  ENDWH
ENDIF
(*----------------------------------------------------------------------------*)
TDISP%:(p%)
REM display to do items on all four lines of the screen,
REM starting at position p% in the file.

LOCAL n%
n%=1
CLS

IF count=0
  printc:("Nothing to do!",2)
  RETURN GET
ELSE
  WHILE n%<4 and (p%+n%)<=count
    at 1,n%+1
    PRINT left$(TDISPIT$:(p%+n%),20)
    n%=n%+1
  ENDWH
  RETURN VIEW(1,TDISPIT$:(p%))
ENDIF
(*----------------------------------------------------------------------------*)
TDISPIT$:(p%)
REM Constructs a one line representation of a todo item

LOCAL d$(5),diff

POSITION p%

IF a.due=99999
  d$="     "
ELSE

  diff=a.due-DAYS(DAY,MONTH,YEAR)
  IF diff<15
    d$=GEN$(diff,-3)+"d "
  ELSE
    d$=fix$(diff/7,0,-3)+"w "
  ENDIF
  
ENDIF

return gen$(a.pr%,1)+d$+a.desc$

DATEIN:(C%,L%,dd%,mm%,yyyy%)
REM Modified from code by David C. Jones
REM Returns DAYS() of selected date or -1 if user escapes
REM using ON/CLEAR

LOCAL E%,G%,DY%,DT%,M%,Y%,CUR%,D$(8),Z$(15),Z(12)
Z(1)=31
Z(3)=31
Z(4)=30
Z(5)=31
Z(6)=30
Z(7)=31
Z(8)=31
Z(9)=30
Z(10)=31
Z(11)=30
Z(12)=31
DT%=dd%
M%=mm%
Y%=yyyy%
DY%=DOW(DT%,M%,Y%)
CUR%=1
CURSOR ON

ST::

Z$=DAYNAME$(DY%)+GEN$(FLT(DT%),-3)+" "+MONTH$(M%)+GEN$(FLT(Y%),-5)
IF (FLT(Y%))/4=INT(FLT(Y%)/4) AND Y%<>1900 AND Y%<>2100
  Z(2)=29
ELSE
  Z(2)=28
ENDIF
AT C%,L%
PRINT Z$
IF CUR%=1
  IF DT%<10
    AT C%+5,L%
  ELSE
    AT C%+4,L%
  ENDIF
ELSEIF CUR%=2
  AT C%+7,L%
ELSEIF CUR%=3
  AT C%+11,L%
ENDIF

CONT::

G%=GET
ONERR ERROR::
IF g%=1
  ONERR OFF
  CURSOR OFF
  RETURN -1
ELSEIF G%=5
  CUR%=CUR%-1
ELSEIF G%=6
  CUR%=CUR%+1
ELSEIF G%=13
  GOTO END::
ELSEIF G%=3 AND CUR%=1
  DT%=DT%+1
ELSEIF G%=4 AND CUR%=1
  DT%=DT%-1
ELSEIF G%=3 AND CUR%=2
  M%=M%+1
ELSEIF G%=4 AND CUR%=2
  M%=M%-1
ELSEIF G%=3 AND CUR%=3
  Y%=Y%+1
ELSEIF G%=4 AND CUR%=3
  Y%=Y%-1
ELSE GOTO CONT::
ENDIF
IF CUR%=4
  CUR%=1
 ELSEIF CUR%=0
  CUR%=3
ENDIF
REST::
DY%=DOW(DT%,M%,Y%)
GOTO ST::
END::
ONERR OFF
CURSOR OFF
RETURN DAYS(dt%,m%,y%)

ERROR::
onerr off
E%=ERR

IF E%=194

  batt:
  GOTO END::

ELSEIF E%=247

  IF M%=13
    M%=1
  ELSEIF M%=0
    M%=12
  ENDIF

  IF DT%=29 AND M%=2 AND CUR%=3
    DT%=28
  ENDIF

  IF DT%=Z(M%)+1 AND CUR%=1
    DT%=1
  ELSEIF DT%=0
    DT%=Z(M%)
  ENDIF

  IF Y%=1899
    Y%=2155
  ELSEIF Y%=2156
    Y%=1900
  ENDIF

  IF DT%>Z(M%) AND CUR%=2
    DT%=Z(M%)
  ENDIF

  ONERR ERROR::
  GOTO REST::

ENDIF

DAY%:(ds)
LOCAL j,d,m,y
j = ds+693902
y = INTF((4*j-1)/146097)
j = 4*j-1-146097*y
d = INTF(j/4)
j = INTF((4*d+3)/1461)
d = 4*d+3-1461*j
d = INTF((d+4)/4)
m = INTF((5*d-3)/153)
d = 5*d-3-153*m
return INT((d+5)/5)
(*----------------------------------------------------------------------------*)
MONTH%:(ds)
LOCAL j,d,m,y
j = ds+693902
y = INTF((4*j-1)/146097)
j = 4*j-1-146097*y
d = INTF(j/4)
j = INTF((4*d+3)/1461)
d = 4*d+3-1461*j
d = INTF((d+4)/4)
m = INTF((5*d-3)/153)
IF m<10
  m = m+3
ELSE
  m = m-9
ENDIF

return INT(m)
(*----------------------------------------------------------------------------*)
YEAR%:(ds)
LOCAL j,d,m,y
j = ds+693902
y = INTF((4*j-1)/146097)
j = 4*j-1-146097*y
d = INTF(j/4)
j = INTF((4*d+3)/1461)
d = 4*d+3-1461*j
d = INTF((d+4)/4)
m = INTF((5*d-3)/153)
y = 100*y+j
IF m>9
  y = y+1
ENDIF

return INT(y)

ZERPREF$:(n%,l%)
local t$(255)
t$=num$(n%,l%)
if len(t$)<l%
 t$=rept$("0",l%-len(t$))+t$
endif
return t$
